<script>
    document.addEventListener("DOMContentLoaded", () => {
        const vocales = ["A", "E", "I", "O", "U"];
        let letraObjetivo = "";
        let puntos = 0;
        let tiempo = 20;
        let intervaloTiempo, intervaloBurbujas;
        let juegoEnPausa = false;

        const musicaFondo = document.getElementById('musicaFondo'); 

        const pantallaInicio = document.getElementById("pantallaInicio");
        const btnComenzar = document.getElementById("btnComenzar");
        const instrucciones = document.getElementById("instrucciones");
        const objetivo = document.getElementById("objetivo");
        const tiempoEl = document.getElementById("tiempo");
        const puntosEl = document.getElementById("puntos");
        const zonaJuego = document.getElementById("zonaJuego");
        const mensajeFinal = document.getElementById("mensajeFinal");
        const btnPausa = document.getElementById("btnPausa");
        const menuPausa = document.getElementById("menuPausa");
        const btnSeguir = document.getElementById("btnSeguir");
        const btnReiniciar = document.getElementById("btnReiniciar");
        const efectoMagico = document.getElementById("efectoMagico");

        // Enlazar los eventos de los botones
        btnComenzar.addEventListener("click", iniciarJuego);
        btnPausa.addEventListener("click", pausarJuego); 
        btnSeguir.addEventListener("click", seguirJuego);
        btnReiniciar.addEventListener("click", volverAlInicio);

        function iniciarJuego() {
            // CORRECCIÓN CLAVE: Iniciar música y voz SÓLO después del clic del usuario
            musicaFondo.play().catch(e => console.log("Música no pudo iniciar automáticamente."));
            
            puntos = 0;
            puntosEl.textContent = puntos;
            pantallaInicio.style.display = "none";
            instrucciones.style.display = "block";
            zonaJuego.style.display = "block";
            btnPausa.style.display = "block";
            menuPausa.style.display = "none"; 
            juegoEnPausa = false;
            nuevaRonda();
        }

        function nuevaRonda() {
            zonaJuego.innerHTML = "";
            mensajeFinal.style.display = "none";
            tiempo = 20;
            tiempoEl.textContent = tiempo;

            letraObjetivo = vocales[Math.floor(Math.random() * vocales.length)];
            objetivo.textContent = letraObjetivo;

            // Opción de síntesis de voz (se ejecuta después del clic inicial)
            if ('speechSynthesis' in window) {
                if (!juegoEnPausa) {
                    const mensaje = new SpeechSynthesisUtterance("Toca la letra " + letraObjetivo);
                    mensaje.lang = "es-ES";
                    speechSynthesis.speak(mensaje);
                }
            }

            clearInterval(intervaloTiempo);
            intervaloTiempo = setInterval(() => {
                if (!juegoEnPausa) {
                    tiempo--;
                    tiempoEl.textContent = tiempo;
                    if (tiempo <= 0) {
                        clearInterval(intervaloTiempo);
                        clearInterval(intervaloBurbujas);
                        finDeRonda();
                    }
                }
            }, 1000);

            clearInterval(intervaloBurbujas);
            intervaloBurbujas = setInterval(() => {
                if (!juegoEnPausa) crearBurbuja();
            }, 1000);
        }

        function colorAleatorio() {
            const colores = ["#ff9999", "#99ff99", "#9999ff", "#ffff99", "#ffccff"];
            return colores[Math.floor(Math.random() * colores.length)];
        }

        function crearBurbuja() {
            const burbuja = document.createElement("div");
            burbuja.className = "burbuja";

            let letra;
            if (Math.random() < 0.4) { 
                letra = letraObjetivo;
            } else {
                do {
                    letra = vocales[Math.floor(Math.random() * vocales.length)];
                } while (letra === letraObjetivo);
            }
            burbuja.textContent = letra;

            burbuja.style.setProperty('--bubble-color', colorAleatorio());
            burbuja.style.left = Math.random() * (window.innerWidth - 120) + "px"; 
            burbuja.style.top = Math.random() * (window.innerHeight - 120) + "px";

            burbuja.addEventListener("click", function handler() {
                burbuja.removeEventListener("click", handler);
                if (letra === letraObjetivo) {
                    puntos++;
                    puntosEl.textContent = puntos;
                    mostrarEfectoMagico(burbuja.getBoundingClientRect());
                } else {
                    tiempo = Math.max(0, tiempo - 2); 
                }
                burbuja.classList.add("explotar");
                setTimeout(() => burbuja.remove(), 200);
            });

            zonaJuego.appendChild(burbuja);
            setTimeout(() => {
                if (zonaJuego.contains(burbuja) && !burbuja.classList.contains("explotar")) {
                    burbuja.remove();
                }
            }, 7000); 
        }

        function mostrarEfectoMagico(rect) {
            efectoMagico.style.display = "block";
            efectoMagico.style.left = (rect.left + rect.width / 2 - 20) + "px";
            efectoMagico.style.top = (rect.top + rect.height / 2 - 20) + "px";

            efectoMagico.classList.remove("brillar");
            void efectoMagico.offsetWidth;
            efectoMagico.classList.add("brillar");

            setTimeout(() => {
                efectoMagico.style.display = "none";
            }, 700);
        }

        function finDeRonda() {
            let record = localStorage.getItem("record") || 0;
            if (puntos > record) {
                localStorage.setItem("record", puntos);
                record = puntos;
                mensajeFinal.textContent = `🎉 ¡NUEVO RÉCORD! ${puntos} Puntos! 🏆`;
            } else {
                mensajeFinal.textContent = `⏰ ¡Tiempo terminado! ⭐ Puntos: ${puntos} 🏆 Récord: ${record}`;
            }
            mensajeFinal.style.display = "block";
            setTimeout(nuevaRonda, 3500); 
        }

        function pausarJuego() {
            juegoEnPausa = true;
            menuPausa.style.display = "flex";
            clearInterval(intervaloTiempo);
            clearInterval(intervaloBurbujas);
            musicaFondo.pause(); 
        }

        function seguirJuego() {
            menuPausa.style.display = "none";
            juegoEnPausa = false;
            
            musicaFondo.play().catch(e => console.log("Música no pudo reanudar.")); 
            
            // Reiniciar temporizadores
            intervaloTiempo = setInterval(() => {
                if (!juegoEnPausa) {
                    tiempo--;
                    tiempoEl.textContent = tiempo;
                    if (tiempo <= 0) {
                        clearInterval(intervaloTiempo);
                        clearInterval(intervaloBurbujas);
                        finDeRonda();
                    }
                }
            }, 1000);
            
            intervaloBurbujas = setInterval(() => {
                if (!juegoEnPausa) crearBurbuja();
            }, 1000);
        }

        function volverAlInicio() {
            clearInterval(intervaloTiempo);
            clearInterval(intervaloBurbujas);
            juegoEnPausa = false;
            menuPausa.style.display = "none";
            zonaJuego.style.display = "none";
            instrucciones.style.display = "none";
            btnPausa.style.display = "none";
            mensajeFinal.style.display = "none";
            pantallaInicio.style.display = "flex";
            musicaFondo.pause(); 
            musicaFondo.currentTime = 0; 
        }
    });
</script>
